return pattern.test(email); } // Validate Phone — numbers, 8-15 digits function validatePhone(phone) { var pattern = /^[\d\s\-+]{8,15}$/; return pattern.test(phone.trim()); } // Validate Message — at least 10 characters function validateMessage(msg) { return msg.trim().length >= 10;
